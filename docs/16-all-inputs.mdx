# useForm with All Input Types

This page demonstrates how to use `form.getInputProps` with every common HTML input type.
All examples use `mode: 'uncontrolled'` for best performance.

---

## Text Input

```tsx
import { useForm } from '@headless-form/react';

function Demo() {
  const form = useForm({
    mode: 'uncontrolled',
    initialValues: { name: '' },
    validate: { name: (v) => !v.trim() ? 'Required' : null },
  });

  return (
    <form onSubmit={form.onSubmit(console.log)}>
      <label>
        Name
        <input
          key={form.key('name')}
          {...form.getInputProps('name')}
          type="text"
          placeholder="Your name"
        />
      </label>
      {form.errors.name && <p>{form.errors.name}</p>}
      <button type="submit">Submit</button>
    </form>
  );
}
```

## Checkbox

```tsx

function Demo() {
  const form = useForm({
    mode: 'uncontrolled',

          key={form.key('newsletter')}

        Subscribe to newsletter

      <button type="submit">Submit</button>

  );
}
```

## Checkbox Group (Multi-select)

Use an array field with individual checkboxes to handle multi-select:

```tsx
import { useForm, isNotEmpty } from '@headless-form/react';

const SKILLS = ['React', 'TypeScript', 'Node.js', 'GraphQL', 'CSS'];

function Demo() {
  const form = useForm({
    mode: 'uncontrolled',
    initialValues: { skills: [] as string[] },
    validate: {
      skills: isNotEmpty('Select at least one skill'),
    },
  });

  const handleChange = (skill: string) => {
    const current = form.getValues().skills;
    const next = current.includes(skill)
      ? current.filter((s) => s !== skill)
      : [...current, skill];
    form.setFieldValue('skills', next);
  };

  return (
    <form onSubmit={form.onSubmit(console.log)}>
      <fieldset>
        <legend>Skills</legend>
        {SKILLS.map((skill) => (
          <label key={skill}>
            <input
              type="checkbox"
              checked={form.getValues().skills.includes(skill)}
              onChange={() => handleChange(skill)}
            />
            {skill}
          </label>
        ))}
      </fieldset>
      {form.errors.skills && <p>{form.errors.skills}</p>}
      <button type="submit">Submit</button>
    </form>
  );
}
```

```tsx

  const plans = [
    { value: 'starter', label: 'Starter — Free' },
    { value: 'pro', label: 'Pro — $9/month' },

    <form onSubmit={form.onSubmit(console.log)}>

      <button type="submit">Submit</button>
```

## File Input

File inputs in uncontrolled mode use `onChange` to capture `FileList`:

```tsx
import { useForm, isNotEmpty } from '@headless-form/react';

function Demo() {
  const form = useForm({
    mode: 'uncontrolled',
    initialValues: { file: null as File | null },
    validate: { file: isNotEmpty('Please select a file') },
  });

  return (
    <form onSubmit={form.onSubmit(console.log)}>
      <label>
        Upload File
        <input
          type="file"
          onChange={(e) => {
            form.setFieldValue('file', e.target.files?.[0] ?? null);
          }}
          accept=".pdf,.doc,.docx"
        />
      </label>
      {form.errors.file && <p>{form.errors.file}</p>}
      <button type="submit">Upload</button>
    </form>
  );
}
```

## Complete Form with All Input Types

```tsx
import { useForm, isEmail, isNotEmpty, hasLength, isInRange } from '@headless-form/react';

interface FullFormValues {
  name: string;
  email: string;
  password: string;
  bio: string;
  age: number;
  country: string;
  role: string;
  notifyByEmail: boolean;
  notifyBySms: boolean;
  preferredColor: string;
  satisfactionScore: number;
  startDate: string;
}

function FullForm() {
  const form = useForm<FullFormValues>({
    mode: 'uncontrolled',
    initialValues: {
      name: '',
      email: '',
      password: '',
      bio: '',
      age: 18,
      country: '',
      role: 'user',
      notifyByEmail: true,
      notifyBySms: false,
      preferredColor: '#1971c2',
      satisfactionScore: 5,
      startDate: '',
    },
    validate: {
      name: isNotEmpty('Name is required'),
      email: isEmail('Invalid email'),
      password: hasLength({ min: 8 }, 'Password must be at least 8 characters'),
      bio: hasLength({ max: 200 }, 'Bio must be 200 characters or less'),
      age: isInRange({ min: 16, max: 100 }, 'Age must be between 16 and 100'),
      country: isNotEmpty('Country is required'),
      startDate: isNotEmpty('Start date is required'),
    },
  });

  return (
    <form onSubmit={form.onSubmit(console.log)}>
      <input key={form.key('name')} {...form.getInputProps('name')} type="text" placeholder="Name" />
      {form.errors.name && <p>{form.errors.name}</p>}

      <input key={form.key('email')} {...form.getInputProps('email')} type="email" placeholder="Email" />
      {form.errors.email && <p>{form.errors.email}</p>}

      <input key={form.key('password')} {...form.getInputProps('password')} type="password" placeholder="Password" />
      {form.errors.password && <p>{form.errors.password}</p>}

      <textarea key={form.key('bio')} {...form.getInputProps('bio')} placeholder="Bio" rows={3} />
      {form.errors.bio && <p>{form.errors.bio}</p>}

      <input key={form.key('age')} {...form.getInputProps('age')} type="number" min={0} max={120} />
      {form.errors.age && <p>{form.errors.age}</p>}

      <select key={form.key('country')} {...form.getInputProps('country')}>
        <option value="">Select country</option>
        <option value="us">United States</option>
        <option value="uk">United Kingdom</option>
      </select>
      {form.errors.country && <p>{form.errors.country}</p>}

      <div>
        {['admin', 'user', 'moderator'].map((r) => (
          <label key={r}>
            <input
              key={form.key('role')}
              type="radio"
              {...form.getInputProps('role', { type: 'radio', value: r })}
            />
            {r}
          </label>
        ))}
      </div>

      <label>
        <input
          key={form.key('notifyByEmail')}
          type="checkbox"
          {...form.getInputProps('notifyByEmail', { type: 'checkbox' })}
        />
        Email notifications
      </label>

      <label>
        <input
          key={form.key('notifyBySms')}
          type="checkbox"
          {...form.getInputProps('notifyBySms', { type: 'checkbox' })}
        />
        SMS notifications
      </label>

      <label>
        Color
        <input key={form.key('preferredColor')} {...form.getInputProps('preferredColor')} type="color" />
      </label>

      <label>
        Satisfaction ({form.getValues().satisfactionScore}/10)
        <input
          key={form.key('satisfactionScore')}
          {...form.getInputProps('satisfactionScore')}
          type="range"
          min={1}
          max={10}
          step={1}
        />
      </label>

      <label>
        Start Date
        <input key={form.key('startDate')} {...form.getInputProps('startDate')} type="date" />
      </label>
      {form.errors.startDate && <p>{form.errors.startDate}</p>}

      <button type="submit" disabled={form.submitting}>
        {form.submitting ? 'Submitting...' : 'Submit'}
      </button>
    </form>
  );
}
```

## Angular: All Input Types

In Angular, `HfFieldDirective` automatically handles the correct binding for each input type:

```typescript
import { Component, ChangeDetectionStrategy } from '@angular/core';
import { injectForm, HfFieldDirective } from '@headless-form/angular';
import { isEmail, isNotEmpty } from '@headless-form/angular';

@Component({
  standalone: true,
  imports: [HfFieldDirective],
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <form (ngSubmit)="submit($event)">
      <!-- Text -->
      <input [hfField]="form" path="name" type="text" placeholder="Name" />

      <!-- Email -->
      <input [hfField]="form" path="email" type="email" placeholder="Email" />

      <!-- Password -->
      <input [hfField]="form" path="password" type="password" placeholder="Password" />

      <!-- Number -->
      <input [hfField]="form" path="age" type="number" min="0" max="120" />

      <!-- Textarea (via the directive) -->
      <textarea [hfField]="form" path="bio" rows="3"></textarea>

      <!-- Select -->
      <select [hfField]="form" path="country">
        <option value="">Select country</option>
        <option value="us">United States</option>
        <option value="uk">United Kingdom</option>
      </select>

      <!-- Checkbox -->
      <input [hfField]="form" path="newsletter" type="checkbox" />

      <!-- Radio group -->
      <input [hfField]="form" path="plan" type="radio" value="starter" />
      <input [hfField]="form" path="plan" type="radio" value="pro" />
      <input [hfField]="form" path="plan" type="radio" value="enterprise" />

      <!-- Number (slider) -->
      <input [hfField]="form" path="rating" type="number" />

      <!-- Date -->
      <input [hfField]="form" path="startDate" type="date" />

      <!-- Errors -->
      @if (form.errors()['name']) { <p>{{ form.errors()['name'] }}</p> }
      @if (form.errors()['email']) { <p>{{ form.errors()['email'] }}</p> }

      <button type="submit" [disabled]="form.submitting()">Submit</button>
    </form>
  `,
})
export class AllInputsFormComponent {
  form = injectForm({
    mode: 'uncontrolled',
    initialValues: {
      name: '',
      email: '',
      password: '',
      age: 18,
      bio: '',
      country: '',
      newsletter: false,
      plan: 'starter',
      rating: 5,
      startDate: '',
    },
    validate: {
      name: isNotEmpty('Name is required'),
      email: isEmail('Invalid email'),
    },
  });

  submit(event: Event) {
    this.form.onSubmit((values) => console.log(values))(event);
  }
}
```
