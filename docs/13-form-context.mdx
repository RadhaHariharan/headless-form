# Form Context

Form context lets you access the form object from deeply nested components without
manually passing props. This is useful for complex forms with many sub-components.

---

## React: createFormContext

`createFormContext` returns a tuple of three items:

1. `FormProvider` — a context provider component
2. `useFormContext` — a hook to read the form from context
3. `useForm` — a typed version of `useForm` with the values type pre-applied

```tsx
import { createFormContext } from '@headlesskit/forms-react';

// Define form values type (required)
interface FormValues {
  age: number;
  name: string;
  email: string;
}

// createFormContext returns [FormProvider, useFormContext, useForm]
const [FormProvider, useFormContext, useForm] = createFormContext<FormValues>();
```

### Usage

```tsx
// ContextField.tsx — consumes the form from context
function NameField() {
  const form = useFormContext();

  return (
    <div>
      <label>Name</label>
      <input
        key={form.key('name')}
        {...form.getInputProps('name')}
        placeholder="Your name"
      />
      {form.errors.name && <p style={{ color: 'red' }}>{form.errors.name}</p>}
    </div>
  );
}

// ParentForm.tsx — creates the form and wraps with FormProvider
function ParentForm() {
  const form = useForm({
    mode: 'uncontrolled',
    initialValues: {
      age: 0,
      name: '',
      email: '',
    },
    validate: {
      name: (v) => v.trim().length < 2 ? 'Too short' : null,
      email: (v) => !/^\S+@\S+/.test(v) ? 'Invalid email' : null,
      age: (v) => v < 0 ? 'Invalid age' : null,
    },
  });

  return (
    // Wrap with FormProvider, passing the form instance
    <FormProvider form={form}>
      <form onSubmit={form.onSubmit(console.log)}>
        <NameField />
        {/* EmailField, AgeField, etc. — all read from context */}
        <button type="submit">Submit</button>
      </form>
    </FormProvider>
  );
}
```

## Store Context in a Separate File

Avoid dependency cycles by keeping form context in a dedicated file:

```typescript
// form-context.ts
import { createFormContext } from '@headlesskit/forms-react';

interface UserFormValues {
  firstName: string;
  lastName: string;
  email: string;
  age: number;
}

// Export the three items with descriptive names
export const [UserFormProvider, useUserFormContext, useUserForm] =
  createFormContext<UserFormValues>();
```

Then import from any component:

```tsx
// fields/EmailField.tsx
import { useUserFormContext } from '../form-context';

export function EmailField() {
  const form = useUserFormContext();

  return (
    <div>
      <label>Email</label>
      <input
        key={form.key('email')}
        {...form.getInputProps('email')}
        type="email"
        placeholder="your@email.com"
      />
      {form.errors.email && <p>{form.errors.email}</p>}
    </div>
  );
}
```

```tsx
// fields/NameFields.tsx
import { useUserFormContext } from '../form-context';

export function NameFields() {
  const form = useUserFormContext();

  return (
    <div>
      <input
        key={form.key('firstName')}
        {...form.getInputProps('firstName')}
        placeholder="First name"
      />
      <input
        key={form.key('lastName')}
        {...form.getInputProps('lastName')}
        placeholder="Last name"
      />
    </div>
  );
}
```

```tsx
// UserForm.tsx
import { UserFormProvider, useUserForm } from './form-context';
import { EmailField } from './fields/EmailField';
import { NameFields } from './fields/NameFields';
import { isEmail, isNotEmpty } from '@headlesskit/forms-react';

export function UserForm() {
  const form = useUserForm({
    mode: 'uncontrolled',
    initialValues: {
      firstName: '',
      lastName: '',
      email: '',
      age: 0,
    },
    validate: {
      firstName: isNotEmpty('First name is required'),
      lastName: isNotEmpty('Last name is required'),
      email: isEmail('Invalid email'),
    },
  });

  return (
    <UserFormProvider form={form}>
      <form onSubmit={form.onSubmit(console.log)}>
        <NameFields />
        <EmailField />
        <button type="submit">Create Account</button>
      </form>
    </UserFormProvider>
  );
}
```

## Multiple Contexts

You can create as many independent form contexts as needed:

```typescript
import { createFormContext } from '@headlesskit/forms-react';

interface ShippingFormValues {
  street: string;
  city: string;
  country: string;
}

interface BillingFormValues {
  cardNumber: string;
  expiry: string;
  cvv: string;
}

export const [ShippingFormProvider, useShippingForm, useShipping] =
  createFormContext<ShippingFormValues>();

export const [BillingFormProvider, useBillingForm, useBilling] =
  createFormContext<BillingFormValues>();
```

## Angular: provideForm and injectFormContext

In Angular, use `provideForm` in a component's `providers` array, then `injectFormContext`
in child components to access the form:

```typescript
// form.token.ts — optional: just use provideForm directly
import { injectForm } from '@headlesskit/angular';

interface CheckoutFormValues {
  name: string;
  email: string;
  address: string;
}

export function createCheckoutForm() {
  return injectForm<CheckoutFormValues>({
    mode: 'uncontrolled',
    initialValues: { name: '', email: '', address: '' },
  });
}
```

```typescript
// parent-form.component.ts
import { Component, ChangeDetectionStrategy } from '@angular/core';
import { injectForm, provideForm, HfFieldDirective } from '@headlesskit/angular';
import { isNotEmpty, isEmail } from '@headlesskit/angular';
import { NameFieldComponent } from './name-field.component';
import { EmailFieldComponent } from './email-field.component';

interface CheckoutValues {
  name: string;
  email: string;
  address: string;
}

@Component({
  standalone: true,
  imports: [HfFieldDirective, NameFieldComponent, EmailFieldComponent],
  changeDetection: ChangeDetectionStrategy.OnPush,
  providers: [], // provideForm is called in the constructor body below
  template: `
    <form (ngSubmit)="submit($event)">
      <app-name-field />
      <app-email-field />
      <input [hfField]="form" path="address" placeholder="Address" />
      <button type="submit">Submit</button>
    </form>
  `,
})
export class CheckoutFormComponent {
  form = injectForm<CheckoutValues>({
    mode: 'uncontrolled',
    initialValues: { name: '', email: '', address: '' },
    validate: {
      name: isNotEmpty('Name is required'),
      email: isEmail('Invalid email'),
      address: isNotEmpty('Address is required'),
    },
  });

  constructor() {
    // Make form available to child components via injectFormContext()
    provideForm(this.form);
  }

  submit(event: Event) {
    this.form.onSubmit((values) => console.log(values))(event);
  }
}
```

```typescript
// name-field.component.ts
import { Component, ChangeDetectionStrategy } from '@angular/core';
import { injectFormContext, HfFieldDirective } from '@headlesskit/angular';

@Component({
  standalone: true,
  selector: 'app-name-field',
  imports: [HfFieldDirective],
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <div>
      <label>Name</label>
      <input [hfField]="form" path="name" placeholder="Full name" />
      @if (form.errors()['name']) {
        <p>{{ form.errors()['name'] }}</p>
      }
    </div>
  `,
})
export class NameFieldComponent {
  // Inject the form provided by a parent component
  form = injectFormContext();
}
```

```typescript
// email-field.component.ts
import { Component, ChangeDetectionStrategy } from '@angular/core';
import { injectFormContext, HfFieldDirective } from '@headlesskit/angular';

@Component({
  standalone: true,
  selector: 'app-email-field',
  imports: [HfFieldDirective],
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <div>
      <label>Email</label>
      <input [hfField]="form" path="email" type="email" placeholder="Email" />
      @if (form.errors()['email']) {
        <p>{{ form.errors()['email'] }}</p>
      }
    </div>
  `,
})
export class EmailFieldComponent {
  form = injectFormContext();
}
```

### FORM_CONTEXT_TOKEN

Internally, `provideForm` and `injectFormContext` use the `FORM_CONTEXT_TOKEN` injection token.
You can use it directly for advanced patterns:

```typescript
import { inject } from '@angular/core';
import { FORM_CONTEXT_TOKEN } from '@headlesskit/angular';
import type { FormApi } from '@headlesskit/angular';

// Access with optional injection
const form = inject(FORM_CONTEXT_TOKEN, { optional: true });
if (!form) {
  throw new Error('No form context found — wrap with provideForm()');
}
```

## Error When No Context Is Found

`useFormContext` (React) throws if called outside a `FormProvider`:

```
[headlesskit/useFormContext] No FormProvider found in the component tree.
Wrap your form with <FormProvider form={form}>.
```

`injectFormContext` (Angular) throws if `provideForm` was not called in a parent:

```
[headlesskit/injectFormContext] No provideForm() found in the injection hierarchy.
Add provideForm(form) to a parent component's providers or constructor.
```

## CreateFormContextResult Type

```typescript
import type { CreateFormContextResult } from '@headlesskit/forms-react';

type Result = CreateFormContextResult<FormValues>;
// -> [FormProvider, useFormContext, useForm]
// [0]: React.FC<{ form: UseFormReturnType<FormValues> }>
// [1]: () => UseFormReturnType<FormValues>
// [2]: (options: UseFormOptions<FormValues>) => UseFormReturnType<FormValues>
```
