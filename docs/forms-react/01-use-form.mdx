# useForm

The primary hook that creates and manages form state. It bridges the framework-agnostic
store from `@headlesskit/forms` to React's rendering model via `useSyncExternalStore` —
the exact same hook works for React Native through `@headlesskit/forms-react-native`,
which re-exports it unchanged.

---

## Basic usage

```tsx
import { useForm } from '@headlesskit/forms-react';

const form = useForm({
  mode: 'uncontrolled',
  initialValues: {
    name: '',
    email: '',
  },
});
```

`useForm` is a thin reactive wrapper around `createFormStore` from `@headlesskit/forms` — the
same store, the same API, just bridged into React's render cycle. Reach for `createFormStore`
directly only if you're building your own framework adapter or working outside any UI framework.

---

## Options

`useForm` accepts the same `UseFormOptions` object as `createFormStore`:

| Option | Type | Default | Description |
|--------|------|---------|-------------|
| `mode` | `'controlled' \| 'uncontrolled'` | `'controlled'` | Controls when value changes trigger re-renders. Use `'uncontrolled'` for best performance. |
| `initialValues` | `Partial<Values>` | `{}` | The starting values for the form. |
| `initialErrors` | `FormErrors` | `{}` | Errors to show immediately on mount. |
| `initialTouched` | `Record<string, boolean>` | `{}` | Fields to mark as touched on mount. |
| `initialDirty` | `Record<string, boolean>` | `{}` | Fields to mark as dirty on mount. |
| `validate` | `FormValidateInput` | — | Validation rules object or a whole-form validator function. |
| `validateInputOnChange` | `boolean \| string[]` | `false` | Validate field(s) on every change. |
| `validateInputOnBlur` | `boolean \| string[]` | `false` | Validate field(s) on blur. |
| `validateDebounce` | `number` | `0` | Milliseconds to debounce per-field validation triggered by change/blur. |
| `clearInputErrorOnChange` | `boolean` | `true` | Clear field error when its value changes. |
| `onValuesChange` | `(values, previous) => void` | — | Called on every value change. |
| `transformValues` | `(values) => TransformedValues` | — | Transform values before `onSubmit` delivers them. |
| `enhanceGetInputProps` | `(payload) => Partial<InputProps>` | — | Add extra props to all `getInputProps` results. |
| `onSubmitPreventDefault` | `'always' \| 'never' \| 'validation-failed'` | `'always'` | Controls when `event.preventDefault()` is called. |
| `touchTrigger` | `'change' \| 'focus'` | `'change'` | Which interaction marks a field as touched. |
| `cascadeUpdates` | `boolean` | `false` | Notify `watch` subscribers on ancestor paths. |
| `name` | `string` | — | Register form in global registry for `createFormActions`. |
| `resolveValidationError` | `(error: unknown) => TError` | — | Maps thrown async errors to the error type. |

## Return Value (UseFormReturnType / FormApi)

### Values

```typescript
// Current form values (reactive in controlled mode, stable ref in uncontrolled)
form.values

// Always returns the latest values regardless of mode
form.getValues()

// Returns current initial-values snapshot (used for dirty comparisons)
form.getInitialValues()

// Set initial values (does not change current values)
form.setInitialValues({ name: 'Jane' })

// Initialize once: sets both values and initialValues; idempotent
form.initialize({ name: 'John', email: 'john@example.com' })

// true after initialize() has been called
form.initialized

// Merge partial values into current values
form.setValues({ name: 'Jane' })
form.setValues((current) => ({ ...current, name: current.name.toUpperCase() }))

// Set a single field value
form.setFieldValue('name', 'Jane')
form.setFieldValue('name', (current) => current + '!')

// Reset a single field to its initial value
form.resetField('name')

// Reset entire form (values, errors, touched, dirty, validating)
form.reset()
```

### Errors

```typescript
// Current errors object
form.errors // { 'user.email': 'Invalid email' }

// Replace all errors
form.setErrors({ name: 'Too short', email: 'Invalid' })

// Set a single field error
form.setFieldError('email', 'Invalid email')

// Clear all errors
form.clearErrors()

// Clear a single field error
form.clearFieldError('email')
```

### Validation

```typescript
// Validate all fields — returns sync result or Promise
const result = form.validate()
// result: { hasErrors: boolean, errors: FormErrors }

// Validate a single field
const fieldResult = form.validateField('email')
// fieldResult: { hasError: boolean, error: TError | null }

// Check validity without setting errors
form.isValid()         // true | false | Promise<boolean>
form.isValid('email')  // check single field

// true while any async validation is pending
form.validating

// true if a specific field is validating asynchronously
form.isValidating('email')
```

### Submission

```typescript
// onSubmit returns an event handler for <form onSubmit>
// (on React Native, use getSubmitHandler from @headlesskit/forms-react-native instead —
// it returns a zero-argument function suitable for <Button onPress>)
const handleSubmit = form.onSubmit(
  // Called with transformed values when valid
  (values, event) => {
    console.log(values);
  },
  // Optional: called with errors when invalid
  (errors, values, event) => {
    console.error(errors);
  }
);

// true while an async onSubmit handler is pending
form.submitting

// Manually control submitting state
form.setSubmitting(true)
form.setSubmitting(false)

// onReset: prevents default and calls form.reset()
<form onReset={form.onReset}>
```

### Input Props

```typescript
// Returns props to spread onto an input element
form.getInputProps('name')
// -> { value, onChange, onFocus, onBlur, error, 'data-path': 'name' }

// Uncontrolled mode returns defaultValue instead of value
form.getInputProps('name')
// -> { defaultValue, onChange, onFocus, onBlur, error, 'data-path': 'name' }

// For checkboxes
form.getInputProps('active', { type: 'checkbox' })
// -> { checked, onChange, ... }

// For radio inputs
form.getInputProps('color', { type: 'radio', value: 'red' })
// -> { checked: color === 'red', value: 'red', onChange, ... }

// Get DOM node (SSR-safe, returns null on server)
form.getInputNode('email') // HTMLElement | null

// Stable key for uncontrolled remounting
form.key('name') // string
```

### Status

```typescript
// Is a field (or any field) touched?
form.isTouched('name')  // boolean
form.isTouched()        // any field touched?

// Is a field (or any field) dirty?
form.isDirty('name')    // boolean
form.isDirty()          // any field dirty?

// Read full records
form.getTouched()  // { name: true, ... }
form.getDirty()    // { email: true, ... }

// Set full records
form.setTouched({ name: true })
form.setDirty({ email: true })

// Reset touched / dirty
form.resetTouched()
form.resetDirty()
form.resetDirty({ name: 'John' }) // update dirty baseline
```

### Lists

```typescript
// Append to a list (or insert at index)
form.insertListItem('users', { name: '', email: '' })
form.insertListItem('users', { name: '', email: '' }, 0) // insert at index 0

// Remove from a list
form.removeListItem('users', 0)

// Replace a list item
form.replaceListItem('users', 0, { name: 'Jane', email: 'jane@example.com' })

// Reorder list items
form.reorderListItem('users', { from: 0, to: 2 })
```

### Watch

```typescript
// Watch a field for changes
const unsubscribe = form.watch('email', ({ value, previousValue, touched, dirty }) => {
  console.log('Email changed from', previousValue, 'to', value);
});

// Unsubscribe later
unsubscribe();
```

### Transform

```typescript
// Apply transformValues to current values
form.getTransformedValues()
```

## Full Example

```tsx
import { useForm } from '@headlesskit/forms-react';
import { isEmail, isNotEmpty, hasLength } from '@headlesskit/forms-react';

interface FormValues {
  name: string;
  email: string;
  password: string;
  age: number;
}

function RegistrationForm() {
  const form = useForm<FormValues>({
    mode: 'uncontrolled',
    initialValues: {
      name: '',
      email: '',
      password: '',
      age: 18,
    },
    validate: {
      name: isNotEmpty('Name is required'),
      email: isEmail('Invalid email'),
      password: hasLength({ min: 8 }, 'Password must be at least 8 characters'),
      age: (value) => value < 18 ? 'Must be 18 or older' : null,
    },
    transformValues: (values) => ({
      ...values,
      name: values.name.trim(),
    }),
  });

  return (
    <form onSubmit={form.onSubmit((values) => console.log(values))}>
      <input key={form.key('name')} {...form.getInputProps('name')} placeholder="Name" />
      {form.errors.name && <p>{form.errors.name}</p>}

      <input key={form.key('email')} {...form.getInputProps('email')} type="email" placeholder="Email" />
      {form.errors.email && <p>{form.errors.email}</p>}

      <input key={form.key('password')} {...form.getInputProps('password')} type="password" placeholder="Password" />
      {form.errors.password && <p>{form.errors.password}</p>}

      <input key={form.key('age')} {...form.getInputProps('age')} type="number" />
      {form.errors.age && <p>{form.errors.age}</p>}

      <button type="submit" disabled={form.submitting}>
        {form.submitting ? 'Submitting...' : 'Register'}
      </button>
    </form>
  );
}
```

## TypeScript: UseFormReturnType

```typescript
import type { UseFormReturnType } from '@headlesskit/forms-react';

// Full type with all generics
type MyForm = UseFormReturnType<
  FormValues,        // Values type
  TransformedValues, // Transformed values (defaults to Values)
  string             // Error type (defaults to unknown)
>;

// Get the type of form.values
type Values = MyForm['values'];

// Get the type of errors
type Errors = MyForm['errors'];
```
